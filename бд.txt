--1. Создание базы данных
CREATE DATABASE gpa_db CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE gpa_db;

-- 2. Создание таблиц
CREATE TABLE Workshops (
    Kod_ceha INT NOT NULL,
    Nomer_ceha VARCHAR(50) NOT NULL,
    Naimenovanie_gazoprovoda VARCHAR(255) NOT NULL,
    PRIMARY KEY (Kod_ceha)
) ENGINE=InnoDB;

CREATE TABLE Gpa_types (
    Kod_tipa_GPA INT NOT NULL,
    Nazvanie VARCHAR(255) NOT NULL,
    Proizvoditel VARCHAR(255) NULL,
    Moschnost_MVt DECIMAL(8,2) NULL,
    Inventarnyj_nomer VARCHAR(100) NULL,
    PRIMARY KEY (Kod_tipa_GPA)
) ENGINE=InnoDB;

CREATE TABLE Units (
    Kod_uzla INT NOT NULL,
    Nazvanie_uzla VARCHAR(255) NOT NULL,
    Kod_tipa_GPA INT NOT NULL,
    PRIMARY KEY (Kod_uzla),
    CONSTRAINT fk_units_gpa_type FOREIGN KEY (Kod_tipa_GPA) REFERENCES Gpa_types(Kod_tipa_GPA)
        ON UPDATE CASCADE ON DELETE RESTRICT
) ENGINE=InnoDB;

CREATE TABLE Spare_parts (
    Kod_zapchasti INT NOT NULL,
    Nazvanie VARCHAR(255) NOT NULL,
    Zavodskoj_nomer VARCHAR(100) NULL,
    Kod_uzla INT NOT NULL,
    Proizvoditel VARCHAR(255) NULL,
    PRIMARY KEY (Kod_zapchasti),
    CONSTRAINT fk_spare_parts_units FOREIGN KEY (Kod_uzla) REFERENCES Units(Kod_uzla)
        ON UPDATE CASCADE ON DELETE RESTRICT
) ENGINE=InnoDB;
CREATE TABLE Gpa (
    Kod_GPA INT NOT NULL,
    Zavodskoj_nomer VARCHAR(100) NOT NULL,
    Kod_tipa_GPA INT NOT NULL,
    Kod_ceha INT NOT NULL,
    Data_vvoda DATE NOT NULL,
    Tekuschij_status_GPA VARCHAR(100) NOT NULL DEFAULT 'В работе',
    Tip_dvigatelya VARCHAR(255) NULL,
    Zavodskoj_nomer_dvigatelya VARCHAR(100) NULL,
    Tip_nagnetatelya VARCHAR(255) NULL,
    Zavodskoj_nomer_nagnetatelya VARCHAR(100) NULL,
    Nalichie_modernizirovannyh_uzlov VARCHAR(255) NULL,
    PRIMARY KEY (Kod_GPA),
    UNIQUE KEY gpa_zavodskoj_nomer_unique (Zavodskoj_nomer),
    CONSTRAINT fk_gpa_type FOREIGN KEY (Kod_tipa_GPA) REFERENCES Gpa_types(Kod_tipa_GPA)
        ON UPDATE CASCADE ON DELETE RESTRICT,
    CONSTRAINT fk_gpa_workshop FOREIGN KEY (Kod_ceha) REFERENCES Workshops(Kod_ceha)
        ON UPDATE CASCADE ON DELETE RESTRICT
) ENGINE=InnoDB;

CREATE TABLE Repairs (
    Kod_remonta INT NOT NULL,
    Kod_GPA INT NOT NULL,
    Data_nachala_remonta DATE NOT NULL,
    Data_okonchaniya_remonta DATE NULL,
    Tip_remonta VARCHAR(100) NOT NULL,
    Nomer_zakaza_TORO VARCHAR(100) NULL,
    Vid_remonta VARCHAR(100) NULL,
    Ispolnitel_remonta VARCHAR(255) NULL,
    Status_remonta VARCHAR(255) NULL,
    PRIMARY KEY (Kod_remonta),
    CONSTRAINT fk_repairs_gpa FOREIGN KEY (Kod_GPA) REFERENCES Gpa(Kod_GPA)
        ON UPDATE CASCADE ON DELETE CASCADE
) ENGINE=InnoDB;
CREATE TABLE Gpa_runtime (
    Kod_GPA INT NOT NULL,
    Data_narabotki_GPA DATE NOT NULL,
    Narabotka_za_sutki DECIMAL(8,2) NULL,
    Narabotka_obschaya DECIMAL(8,2) NULL,
    Narabotka_v_rezerve DECIMAL(8,2) NULL,
    Narabotka_v_PPR DECIMAL(8,2) NULL,
    Narabotka_v_vynuzhdennom_prostoe DECIMAL(8,2) NULL,
    Narabotka_posle_SR DECIMAL(8,2) NULL,
    Narabotka_posle_KR DECIMAL(8,2) NULL,
    Narabotka_posle_TR DECIMAL(8,2) NULL,
    Narabotka_po_TO DECIMAL(8,2) NULL,
    Sutochnaya_proizvoditelnost_CBN DECIMAL(8,2) NULL,
    PRIMARY KEY (Kod_GPA, Data_narabotki_GPA),
    CONSTRAINT fk_gpa_runtime_gpa FOREIGN KEY (Kod_GPA) REFERENCES Gpa(Kod_GPA)
        ON UPDATE CASCADE ON DELETE CASCADE
) ENGINE=InnoDB;

CREATE TABLE Failures (
    Kod_GPA INT NOT NULL,
    Data_vremya_otkaza DATETIME NOT NULL,
    Kod_uzla INT NULL,
    Prichina_otkaza TEXT NOT NULL,
    Kategoriya_otkaza VARCHAR(100) NOT NULL,
    Vremya_prostoya DECIMAL(8,2) NOT NULL,
    Mery TEXT NOT NULL,
    Tip_ostanova VARCHAR(100) NOT NULL,
    Nomer_akta_otkaza VARCHAR(100) NOT NULL,
    Data_akta_otkaza DATE NOT NULL,
    PRIMARY KEY (Kod_GPA, Data_vremya_otkaza),
    CONSTRAINT fk_failures_gpa FOREIGN KEY (Kod_GPA) REFERENCES Gpa(Kod_GPA)
        ON UPDATE CASCADE ON DELETE CASCADE,
    CONSTRAINT fk_failures_units FOREIGN KEY (Kod_uzla) REFERENCES Units(Kod_uzla)
        ON UPDATE CASCADE ON DELETE SET NULL
) ENGINE=InnoDB;

CREATE TABLE Installed_parts (
    Kod_ustanovki INT NOT NULL,
    Kod_GPA INT NOT NULL,
    Kod_zapchasti INT NOT NULL,
    Kod_uzla INT NOT NULL,
    Data_ustanovki DATE NOT NULL,
    Narabotka_na_moment_ustanovki DECIMAL(8,2) NOT NULL,
    Prichina_zameny TEXT NOT NULL,
    Kod_remonta INT NULL,
    Status_zapchasti VARCHAR(100) NOT NULL DEFAULT 'Установлена',
    PRIMARY KEY (Kod_ustanovki),
    CONSTRAINT fk_installed_parts_gpa FOREIGN KEY (Kod_GPA) REFERENCES Gpa(Kod_GPA)
        ON UPDATE CASCADE ON DELETE CASCADE,
    CONSTRAINT fk_installed_parts_spare FOREIGN KEY (Kod_zapchasti) REFERENCES Spare_parts(Kod_zapchasti)
        ON UPDATE CASCADE ON DELETE RESTRICT,
    CONSTRAINT fk_installed_parts_units FOREIGN KEY (Kod_uzla) REFERENCES Units(Kod_uzla)
        ON UPDATE CASCADE ON DELETE RESTRICT,
    CONSTRAINT fk_installed_parts_repair FOREIGN KEY (Kod_remonta) REFERENCES Repairs(Kod_remonta)
        ON UPDATE CASCADE ON DELETE SET NULL
) ENGINE=InnoDB;

CREATE TABLE Part_runtime (
    Kod_zapchasti INT NOT NULL,
    Data_narabotki_zapchasti DATE NOT NULL,
    Narabotka_zapchasti_za_sutki DECIMAL(8,2) NULL,
    Narabotka_obschaya_po_zapchasti DECIMAL(8,2) NULL,
    Resurs DECIMAL(8,2) NULL,
    Ostatochnyj_resurs DECIMAL(8,2) NULL,
    Resurs_po_EPB DECIMAL(8,2) NULL,
    Ostatochnyj_resurs_posle_EPB DECIMAL(8,2) NULL,
    PRIMARY KEY (Kod_zapchasti, Data_narabotki_zapchasti),
    CONSTRAINT fk_part_runtime_spare FOREIGN KEY (Kod_zapchasti) REFERENCES Spare_parts(Kod_zapchasti)
        ON UPDATE CASCADE ON DELETE CASCADE
) ENGINE=InnoDB;
CREATE TABLE Problem_issues (
    Kod_problemy INT NOT NULL,
    Kod_GPA INT NOT NULL,
    Kod_uzla INT NOT NULL,
    Kod_zapchasti INT NULL,
    Data_vyyavleniya DATE NOT NULL,
    Opisanie_problemy TEXT NOT NULL,
    Kritichnost VARCHAR(50) NOT NULL,
    Status_problemy VARCHAR(50) NOT NULL DEFAULT 'Открыт',
    Prinyatye_mery TEXT NOT NULL,
    Data_resheniya DATE NULL,
    PRIMARY KEY (Kod_problemy),
    CONSTRAINT fk_problem_issues_gpa FOREIGN KEY (Kod_GPA) REFERENCES Gpa(Kod_GPA)
        ON UPDATE CASCADE ON DELETE CASCADE,
    CONSTRAINT fk_problem_issues_units FOREIGN KEY (Kod_uzla) REFERENCES Units(Kod_uzla)
        ON UPDATE CASCADE ON DELETE RESTRICT,
    CONSTRAINT fk_problem_issues_spare FOREIGN KEY (Kod_zapchasti) REFERENCES Spare_parts(Kod_zapchasti)
        ON UPDATE CASCADE ON DELETE SET NULL
) ENGINE=InnoDB;

CREATE TABLE Safety_expertise (
    Kod_EPB INT NOT NULL,
    Kod_GPA INT NOT NULL,
    Data_provedeniya DATE NULL,
    Nomer_zaklyucheniya VARCHAR(100) NULL,
    Organizaciya VARCHAR(255) NULL,
    Ustanovlennyj_resurs DECIMAL(8,2) NULL,
    Ostatochnyj_resurs DECIMAL(8,2) NULL,
    PRIMARY KEY (Kod_EPB),
    CONSTRAINT fk_safety_expertise_gpa FOREIGN KEY (Kod_GPA) REFERENCES Gpa(Kod_GPA)
        ON UPDATE CASCADE ON DELETE CASCADE
) ENGINE=InnoDB;

CREATE TABLE Forced_idle (
    Kod_prostoya INT NOT NULL,
    Kod_GPA INT NOT NULL,
    Prichina VARCHAR(255) NULL,
    Data_nachala_prostoya DATE NULL,
    Data_okonchaniya DATE NULL,
    Vremya_prostoya DECIMAL(8,2) NULL,
    PRIMARY KEY (Kod_prostoya),
    CONSTRAINT fk_forced_idle_gpa FOREIGN KEY (Kod_GPA) REFERENCES Gpa(Kod_GPA)
        ON UPDATE CASCADE ON DELETE CASCADE
) ENGINE=InnoDB;

CREATE TABLE Parts_movement (
    Kod_dvizheniya INT NOT NULL,
    Kod_remonta INT NULL,
    Kod_zapchasti INT NOT NULL,
    Planovaya_data_postavki DATE NOT NULL,
    Fakticheskaya_data_postavki DATE NOT NULL,
    Data_vhodnogo_kontrolya DATE NOT NULL,
    Nomer_akta_VK VARCHAR(100) NOT NULL,
    Nomer_nakladnoj VARCHAR(100) NOT NULL,
    Data_nakladnoj DATE NOT NULL,
    Nomer_i_data_pasporta VARCHAR(255) NOT NULL,
    Mestopolozhenie VARCHAR(255) NOT NULL,
    Stoimost DECIMAL(10,2) NOT NULL,
    Spisanie VARCHAR(255) NOT NULL,
    PRIMARY KEY (Kod_dvizheniya),
    CONSTRAINT fk_parts_movement_repair FOREIGN KEY (Kod_remonta) REFERENCES Repairs(Kod_remonta)
        ON UPDATE CASCADE ON DELETE SET NULL,
    CONSTRAINT fk_parts_movement_spare FOREIGN KEY (Kod_zapchasti) REFERENCES Spare_parts(Kod_zapchasti)
        ON UPDATE CASCADE ON DELETE RESTRICT
) ENGINE=InnoDB;

CREATE TABLE Starts (
    Kod_puska INT NOT NULL,
    Kod_GPA INT NOT NULL,
    Data_puska DATE NULL,
    PRIMARY KEY (Kod_puska),
    CONSTRAINT fk_starts_gpa FOREIGN KEY (Kod_GPA) REFERENCES Gpa(Kod_GPA)
        ON UPDATE CASCADE ON DELETE CASCADE
) ENGINE=InnoDB;

CREATE TABLE Shutdowns (
    Kod_ostanova INT NOT NULL,
    Kod_GPA INT NOT NULL,
    Tip_ostanova VARCHAR(100) NULL,
    Data_ostanova DATE NULL,
    PRIMARY KEY (Kod_ostanova),
    CONSTRAINT fk_shutdowns_gpa FOREIGN KEY (Kod_GPA) REFERENCES Gpa(Kod_GPA)
        ON UPDATE CASCADE ON DELETE CASCADE
) ENGINE=InnoDB;

CREATE TABLE Oil_consumption (
    Kod_raskhoda INT NOT NULL,
    Kod_GPA INT NOT NULL,
    Data_raskhoda DATE NULL,
    Sutochnyj_raskhod_kg_ch DECIMAL(8,2) NULL,
    PRIMARY KEY (Kod_raskhoda),
    CONSTRAINT fk_oil_consumption_gpa FOREIGN KEY (Kod_GPA) REFERENCES Gpa(Kod_GPA)
        ON UPDATE CASCADE ON DELETE CASCADE
) ENGINE=InnoDB;

CREATE TABLE Form_ks2 (
    Kod_zapisi INT NOT NULL,
    Kod_GPA INT NOT NULL,
    Kod_remonta INT NULL,
    Data_sostavleniya DATE NOT NULL,
    Zatraty_na_SMR DECIMAL(10,2) NOT NULL,
    Zatraty_na_MTR DECIMAL(10,2) NOT NULL,
    Perechen_vovlechennogo_MTR TEXT NOT NULL,
    Podryadnaya_organizaciya VARCHAR(255) NOT NULL,
    PRIMARY KEY (Kod_zapisi),
    CONSTRAINT fk_form_ks2_gpa FOREIGN KEY (Kod_GPA) REFERENCES Gpa(Kod_GPA)
        ON UPDATE CASCADE ON DELETE CASCADE,
    CONSTRAINT fk_form_ks2_repair FOREIGN KEY (Kod_remonta) REFERENCES Repairs(Kod_remonta)
        ON UPDATE CASCADE ON DELETE SET NULL
) ENGINE=InnoDB;

CREATE TABLE Diagnostics (
    Kod_diagnostiki INT NOT NULL,
    Kod_GPA INT NOT NULL,
    Vid_diagnostiki VARCHAR(255) NULL,
    Rezultaty_DO TEXT NULL,
    Data_provedeniya_DO DATE NULL,
    Status_diagnostiki VARCHAR(100) NULL,
    Primechanie TEXT NULL,
    PRIMARY KEY (Kod_diagnostiki),
    CONSTRAINT fk_diagnostics_gpa FOREIGN KEY (Kod_GPA) REFERENCES Gpa(Kod_GPA)
        ON UPDATE CASCADE ON DELETE CASCADE
) ENGINE=InnoDB;

CREATE TABLE Heat_recovery_units (
    Kod_utilizatora INT NOT NULL,
    Kod_GPA INT NOT NULL,
    Nalichie VARCHAR(100) NULL,
    Vid_utilizatora VARCHAR(100) NULL,
    Status_utilizatora VARCHAR(100) NULL,
    Tip_utilizatora VARCHAR(100) NULL,
    PRIMARY KEY (Kod_utilizatora),
    CONSTRAINT fk_heat_recovery_units_gpa FOREIGN KEY (Kod_GPA) REFERENCES Gpa(Kod_GPA)
        ON UPDATE CASCADE ON DELETE CASCADE
) ENGINE=InnoDB;

-- Часть 2
-- 1. Неявное соединение
-- Вывод списка ГПА, которые находятся в статусе «В работе» или «Резерв», с указанием их типа, мощности и названия цеха. Отсортировать по мощности (по убыванию).
SELECT 
    g.Kod_GPA, 
    g.Zavodskoj_nomer, 
    t.Nazvanie AS Tip_GPA, 
    t.Moschnost_MVt, 
    w.Nomer_ceha,
    g.Tekuschij_status_GPA
FROM 
    Gpa g, Gpa_types t, Workshops w
WHERE 
    g.Kod_tipa_GPA = t.Kod_tipa_GPA 
    AND g.Kod_ceha = w.Kod_ceha
    AND g.Tekuschij_status_GPA IN ('В работе', 'Резерв')
ORDER BY 
    t.Moschnost_MVt DESC;


-- Вывод диагностик, которые проводились "внепланово" или "после ремонта"
SELECT 
    d.Kod_diagnostiki,
    g.Zavodskoj_nomer AS GPA,
    d.Vid_diagnostiki,
    d.Status_diagnostiki,
    d.Data_provedeniya_DO
FROM 
    Diagnostics d, Gpa g
WHERE 
    d.Kod_GPA = g.Kod_GPA
    AND d.Status_diagnostiki IN ('внеплановая', 'после ремонта')
ORDER BY 
    d.Data_provedeniya_DO DESC;

-- 2. Явное соединение 
--Вывод истории ремонтов с указанием заводского номера ГПА и номера цеха, где он находится. 
SELECT 
    r.Kod_remonta,
    r.Tip_remonta,
    r.Data_nachala_remonta,
    r.Data_okonchaniya_remonta,
    g.Zavodskoj_nomer AS GPA_Nomer,
    w.Nomer_ceha
FROM 
    Repairs r
INNER JOIN Gpa g ON r.Kod_GPA = g.Kod_GPA
INNER JOIN Workshops w ON g.Kod_ceha = w.Kod_ceha
ORDER BY 
    r.Data_nachala_remonta DESC;

-- Вывод всех проблемных вопросов с указанием узлов, к которым они относятся
SELECT 
    p.Kod_problemy,
    g.Zavodskoj_nomer AS GPA,
    u.Nazvanie_uzla,
    p.Opisanie_problemy,
    p.Kritichnost
FROM 
    Problem_issues p
INNER JOIN Gpa g ON p.Kod_GPA = g.Kod_GPA
INNER JOIN Units u ON p.Kod_uzla = u.Kod_uzla;

-- 3. Агрегирующие операции
-- Вывод общей статистики по наработке ГПА: сумму общей наработки, среднюю суточную наработку, а также максимальную и минимальную наработку в резерве.
SELECT 
    COUNT(*) AS Kolichestvo_zapisej,
    SUM(Narabotka_obschaya) AS Summarnaya_obschaya_narabotka,
    ROUND(AVG(Narabotka_za_sutki), 2) AS Srednyaya_narabotka_za_sutki,
    MAX(Narabotka_v_rezerve) AS Max_narabotka_v_rezerve,
    MIN(Narabotka_v_rezerve) AS Min_narabotka_v_rezerve
FROM 
    Gpa_runtime;
-- Вывод общей стоимости всех закупленных запчастей и средней цены одной запчасти
SELECT 
    COUNT(*) AS Kolichestvo_postavok,
    SUM(Stoimost) AS Obshaya_stoimost,
    ROUND(AVG(Stoimost), 2) AS Srednyaya_stoimost,
    MAX(Stoimost) AS Samaya_dorogaya,
    MIN(Stoimost) AS Samaya_deshevaya
FROM 
    Parts_movement;

-- Вывод максимального и минимального времени простоя ГПА
SELECT 
    MAX(Vremya_prostoya) AS Max_prostoy,
    MIN(Vremya_prostoya) AS Min_prostoy,
    ROUND(AVG(Vremya_prostoya), 2) AS Sredniy_prostoy
FROM 
    Failures;

-- 4. Группировка и фильтрация 
-- Вывод кодов ГПА и количество установленных на них запчастей. Оставить только те ГПА, на которых установлено 2 и более запчастей
SELECT 
    ip.Kod_GPA,
    g.Zavodskoj_nomer,
    COUNT(ip.Kod_ustanovki) AS Kolichestvo_ustanovlennyh_zapchastej
FROM 
    Installed_parts ip
INNER JOIN Gpa g ON ip.Kod_GPA = g.Kod_GPA
GROUP BY 
    ip.Kod_GPA, g.Zavodskoj_nomer
HAVING 
    COUNT(ip.Kod_ustanovki) > 1
ORDER BY 
    Kolichestvo_ustanovlennyh_zapchastej DESC;

 -- Считает, сколько денег потратили на СМР и МТР по каждому ГПА.
SELECT 
    f.Kod_GPA,
    g.Zavodskoj_nomer,
    SUM(f.Zatraty_na_SMR) AS Itogo_SMR,
    SUM(f.Zatraty_na_MTR) AS Itogo_MTR,
    SUM(f.Zatraty_na_SMR + f.Zatraty_na_MTR) AS Obshie_zatraty
FROM 
    Form_ks2 f
JOIN Gpa g ON f.Kod_GPA = g.Kod_GPA
GROUP BY 
    f.Kod_GPA, g.Zavodskoj_nomer
ORDER BY 
    Obshie_zatraty DESC;

-- 5. Вложенные подзапросы
-- Найти все детали (запчасти), которые когда-либо устанавливались на ГПА, находящиеся в цехе с номером '1'
SELECT Kod_zapchasti, Nazvanie, Proizvoditel
FROM Spare_parts
WHERE Kod_uzla IN (
    SELECT Kod_uzla 
    FROM Units 
    WHERE Kod_tipa_GPA IN (
        SELECT Kod_tipa_GPA 
        FROM Gpa 
        WHERE Kod_ceha = 1 -- Код цеха №1
    )
);

-- Вывести информацию о самой мощной установке
SELECT Kod_GPA, Zavodskoj_nomer, Kod_tipa_GPA
FROM Gpa
WHERE Kod_tipa_GPA = ALL (
    SELECT Kod_tipa_GPA 
    FROM Gpa_types 
    WHERE Moschnost_MVt >= 16.0
);

-- 6. Функции управления потоком 
-- Вывести список ГПА и их статус. Если статус «В работе», вывести «Активен», если «Резерв» — «В запасе», а если «На ремонте» — вывести текст с предупреждением
SELECT 
    Kod_GPA,
    Zavodskoj_nomer,
    Tekuschij_status_GPA,
    CASE 
        WHEN Tekuschij_status_GPA = 'В работе' THEN 'Активен'
        WHEN Tekuschij_status_GPA = 'Резерв' THEN 'В запасе'
        WHEN Tekuschij_status_GPA = 'На ремонте' THEN 'Требует внимания!'
        ELSE 'Неизвестный статус'
    END AS Status_opisanie
FROM 
    Gpa;

-- Оценка остаточного ресурса запчасти. Если остаток меньше 5000, вывести «Критический», иначе «Норма»

    Ostatochnyj_resurs,
    IF(Ostatochnyj_resurs < 5000, 'Критический износ', 'Норма') AS Sostoyanie
